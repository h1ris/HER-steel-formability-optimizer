import streamlit as st
import pandas as pd
import numpy as np
import joblib

st.set_page_config(page_title="Automotive Steel HER Dashboard", layout="wide")

# Safe loading wrapper for the unified machine learning pipeline artifact
@st.cache_resource
def load_pipeline():
    return joblib.load('final_her_pipeline.joblib')

try:
    pipeline = load_pipeline()
    model_loaded = True
except Exception as e:
    st.error(f"Configuration Error: 'final_her_pipeline.joblib' not detected in the current active folder path.")
    st.info("Ensure the downloaded file from Kaggle is saved exactly alongside this script.")
    model_loaded = False

st.title("⚙️ Automotive DP/CP Steel Edge Formability Optimizer")
st.markdown("Predict and optimize the **Hole Expansion Ratio (HER %)** for advanced automotive sheet engineering components.")

if model_loaded:
    mode = st.sidebar.radio("Application Operation Mode:", ["Forward Predictor Engine", "Reverse Process Optimizer"])

    if mode == "Forward Predictor Engine":
        st.subheader("🔮 Mode 1: Forward Performance Prediction")
        col1, col2, col3 = st.columns(3)
        
        with col1:
            st.markdown("### Process Geometry")
            # Clear Dropdown Selectors for Categorical Attributes
            steel_grade = st.selectbox("Steel Designation Grade", ["DP600", "DP780", "DP800", "DP980", "DP1180", "CP590", "CP800", "CP1200"])
            steel_type = "CP" if "CP" in steel_grade else "DP"
            
            hole_prep = st.selectbox("Hole Blanking Methodology", ["Sheared", "Reamed"])
            burr_ori = "None" if hole_prep == "Reamed" else st.selectbox("Resulting Burr Orientation", ["Burr Down", "Burr Up"])
            punch_geo = st.selectbox("Stamping Punch Geometry Type", ["Conical", "Flat"])

        with col2:
            st.markdown("### Numerical Dimensions")
            # Pure Numerical Inputs replacing sliders for exact physical accuracy
            thickness = st.number_input("Sheet Thickness (mm)", min_value=0.5, max_value=6.0, value=2.5, step=0.1)
            clearance = st.number_input("Die Tooling Clearance (%)", min_value=2.0, max_value=50.0, value=12.0, step=0.5)

        with col3:
            st.markdown("### Material Tensile Values")
            # Smart baseline defaults depending on the user's chosen grade profile
            default_ys = 548 if "CP" in steel_grade else 450
            default_uts = 614 if "CP" in steel_grade else 600
            default_n = 0.10 if "CP" in steel_grade else 0.15
            
            ys = st.number_input("Yield Strength, YS (MPa)", min_value=100, max_value=1500, value=default_ys, step=1)
            uts = st.number_input("Ultimate Tensile Strength, UTS (MPa)", min_value=200, max_value=2000, value=default_uts, step=1)
            n_val = st.number_input("Strain Hardening Exponent (n-value)", min_value=0.01, max_value=0.40, value=default_n, step=0.01)

        if st.button("Run Prediction Engine"):
            # Execute automated macro mechanical property feature engine mapping
            psm = (uts - ys) * n_val
            s_ratio = ys / uts
            uts_n = uts * n_val
            
            payload = pd.DataFrame([{
                'Steel': steel_grade, 'Type': steel_type, 'Thickness_mm': thickness,
                'YS_MPa': ys, 'UTS_MPa': uts, 'n_value': n_val, 'Clearance_pct': clearance,
                'Hole_Preparation': hole_prep, 'Burr_Orientation': burr_ori, 'Punch_Geometry': punch_geo,
                'Plastic_Strain_Margin': psm, 'Strength_Ratio_YS_UTS': s_ratio, 'UTS_x_n': uts_n
            }])
            
            # Compute end-to-end forward vector processing arrays
            prediction = pipeline.predict(payload)[0]
            
            st.balloons()
            st.success(f"### Predicted Hole Expansion Ratio (HER): {prediction:.2f}%")

    elif mode == "Reverse Process Optimizer":
        st.subheader("🛠️ Mode 2: Inverse Tooling Clearance Constraint Search")
        
        col1, col2 = st.columns(2)
        with col1:
            target_her = st.number_input("Minimum Desired Edge HER Target (%)", min_value=20, max_value=250, value=90, step=5)
            steel_grade = st.selectbox("Target Material Grade Selection", ["DP600", "DP780", "DP800", "CP590"])
            steel_type = "CP" if "CP" in steel_grade else "DP"
        with col2:
            thickness = st.number_input("Target Design Sheet Thickness (mm)", min_value=0.5, max_value=5.0, value=2.0, step=0.1)
            punch_geo = st.selectbox("Preferred Punch Profile Type", ["Conical", "Flat"])

        if st.button("Locate Recommended Tool Clearances"):
            clearance_space = np.linspace(5, 35, 60)
            viable_results = []
            
            ys = 548 if steel_type == "CP" else 450
            uts = 614 if steel_type == "CP" else 600
            n_val = 0.10 if steel_type == "CP" else 0.15
            psm = (uts - ys) * n_val
            
            # Scan structural boundary combinations across the simulated parameter arrays
            for c in clearance_space:
                for b_ori in ["Burr Down", "Burr Up"]:
                    payload = pd.DataFrame([{
                        'Steel': steel_grade, 'Type': steel_type, 'Thickness_mm': thickness,
                        'YS_MPa': ys, 'UTS_MPa': uts, 'n_value': n_val, 'Clearance_pct': c,
                        'Hole_Preparation': "Sheared", 'Burr_Orientation': b_ori, 'Punch_Geometry': punch_geo,
                        'Plastic_Strain_Margin': psm, 'Strength_Ratio_YS_UTS': ys/uts, 'UTS_x_n': uts*n_val
                    }])
                    pred = pipeline.predict(payload)[0]
                    if pred >= target_her:
                        viable_results.append((c, b_ori, pred))
                        
            if viable_results:
                best_config = sorted(viable_results, key=lambda x: x[2], reverse=True)[0]
                st.success(f"### Optimal Configuration Isolated!")
                st.markdown(f"""
                To reliably achieve an edge expandability target above **{target_her}%**:
                * **Recommended Die Clearance:** `{best_config[0]:.1f}%`
                * **Required Blanking Orientation:** `{best_config[1]}` 
                * **Expected Resulting HER Outcome:** `{best_config[2]:.2f}%`
                """)
            else:
                st.warning("No configuration satisfies this target matrix with nominal grade properties. Consider adjusting the targeted performance parameter.")
